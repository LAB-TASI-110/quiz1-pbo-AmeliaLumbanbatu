 package model;

import java.util.ArrayList;
import java.util.List;

public class JadwalLaundry {
    private String nama;
    private String asrama;
    private double berat;
    private String hariAmbil;
    private List<String> isiBag;

    public JadwalLaundry(String nama, String asrama, double berat, String hariAmbil) {
        this.nama = nama;
        this.asrama = asrama;
        this.berat = berat;
        this.hariAmbil = hariAmbil;
        this.isiBag = new ArrayList<>();
    }

    public void tambahItem(String item) {
        this.isiBag.add(item);
    }

    public String getEstimasiKembali() {
        // Logika H+1 Sore sesuai permintaanmu
        switch (hariAmbil.toLowerCase()) {
            case "senin": return "Selasa Sore";
            case "selasa": return "Rabu Sore";
            case "rabu": return "Kamis Sore";
            case "jumat": return "Sabtu Sore";
            default: return "Tidak ada jadwal";
        }
    }

    public void tampilkanDetail() {
        System.out.println("\n--- DATA OPERASIONAL LAUNDRY DEL ---");
        System.out.println("Nama Mahasiswa : " + nama);
        System.out.println("Asrama         : " + asrama);
        System.out.println("Berat          : " + berat + " kg " + (berat > 7 ? "(MELEBIHI KUOTA 7KG!)" : "(Aman)"));
        System.out.println("Isi Bag        : " + isiBag);
        System.out.println("Jadwal Kembali : " + getEstimasiKembali());
        System.out.println("-------------------------------------");
    }
}
